# 示例：一条主张怎么过闸

```bash
python3 src/gate.py init --vault demo
python3 src/gate.py --vault demo entity add E01 "某客户" --note "外部联系人"

cat > /tmp/c1.json <<'JSON'
{"layer":"L2","subject":"E01","relation":"认为",
 "object":"报价前先确认交付物格式","dimension":"销售流程",
 "valid_from":"2026-11-06","source":"会议","source_ref":"inbox/20261106.md#L8",
 "level":"V1","level_by":"agent",
 "verify":"按此流程走一单，看是否少返工一次",
 "use_when":"接新客户报价时"}
JSON

python3 src/gate.py --vault demo add --json /tmp/c1.json
# => {"ok": true, "status": "active", "id": "C-...001", "where": "claims"}

# 再交一次同一份 → 被拒（不是报错，是让你去改那一条）
python3 src/gate.py --vault demo add --json /tmp/c1.json
# => {"ok": false, "status": "duplicate", "existing": "C-...001"}

# 写一条没写 use_when 的 → 进候选区
sed 's/"use_when":"接新客户报价时"//' /tmp/c1.json | sed 's/"object":"报价前先确认交付物格式"/"object":"另一个结论"/' > /tmp/c2.json
python3 src/gate.py --vault demo add --json /tmp/c2.json
# => {"ok": true, "status": "candidate", "why": "缺 use_when…"}
```
